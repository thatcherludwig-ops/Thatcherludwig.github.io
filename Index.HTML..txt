<script>
const searchBox = document.getElementById("searchBox");

searchBox.addEventListener("keyup", function(){
    let filter = this.value.toLowerCase();
    let recipes = document.querySelectorAll(".recipe");

    recipes.forEach(recipe => {
        let text = recipe.innerText.toLowerCase();
        recipe.style.display = text.includes(filter)
            ? "block"
            : "none";
    });
});

function showRecipe(type){

if(type==="brioche"){
alert(
`Brioche Bread

Ingredients:
4 cups flour
2 tsp yeast
1/3 cup sugar
4 eggs
1/2 cup butter

Mix, knead, rise, shape and bake at 375°F for 25-30 min.`
);
}

if(type==="croissant"){
alert(
`Croissants

Laminated dough with butter folds.
Bake at 400°F for 18-22 min.`
);
}

if(type==="lemon"){
alert(
`Lemon Loaf

2 cups flour
1 cup sugar
2 eggs
1/2 cup milk
Lemon zest

Bake at 350°F for 45-50 min.`
);
}
}
